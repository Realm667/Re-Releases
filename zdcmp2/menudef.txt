//-------------------------------------------------------------------------------------------
//
// ZDCMP2 Modded menus -- used to display Jimmyfont using text instead of static patches.
//
//-------------------------------------------------------------------------------------------

DefaultListMenu
{
	Font "BigFont", "Untranslated"
	Selector "M_SKULL1", -32, -5
	Linespacing 16
}

ListMenu "MainMenu"
{
    StaticPatch 94, 12, "M_DOOM"
    Position 97, 77

    TextItem "$MNU_NEWGAME", "n", "PlayerclassMenu"
    TextItem "$MNU_LOADGAME", "l", "LoadGameMenu"
    TextItem "$MNU_SAVEGAME", "s", "SaveGameMenu"
    TextItem "$ZDC2_OPTIONS", "z", "ZDCMP2Menu"
    TextItem "$MNU_OPTIONS", "o", "OptionsMenu"
    TextItem "$MNU_QUITGAME", "q", "QuitMenu"
}

// The built-in menu templates fill the episode and skill entries at runtime.
// Only their static header graphics are replaced here.
ListMenu "EpisodeMenu"
{
    NetgameMessage "$NEWGAME"
    StaticTextCentered 160, 38, "$MNU_EPISODE"
    Position 48, 63
    LineSpacing 16
}

ListMenu "SkillMenu"
{
    NetgameMessage "$NEWGAME"
    StaticTextCentered 160, 14, "$MNU_NEWGAME"
    StaticTextCentered 160, 38, "$MNU_CHOOSESKILL"
    Position 48, 63
    LineSpacing 16
}

	StaticText 	""

	Option      "$ZDC2_AUTOSAVES"           , "zdcmp2_autosave", "YesNo"
	Option      "$ZDC2_CREDITS"        , "zdcmp2_credits",  "YesNo"
	Option      "$ZDC2_EFFECTS" , "zdcmp2_effects",  "YesNo"
	StaticText 	""
	StaticText 	"$ZDC2_CONTROLS"              , 1
	Control		"$ZDC2_THROW"                , "+zdcmp2grenade"
	Control 	"$ZDC2_OBJECTIVES"      , "showpop 3"
	Control 	"$ZDC2_LOG"             , "showpop 1"
	Control 	"$ZDC2_PREV_LOG"     , "zdcmp2prevLog"
	Control 	"$ZDC2_NEXT_LOG"         , "zdcmp2nextLog"
