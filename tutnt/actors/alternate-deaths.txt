// Alternate death states from the four TNT releases; see _docs/utnt/UTNT_RELEASE_EFFECTS.md.

// Narrow replacement chains preserve UTNT combat classes and their custom descendants.

// Death selection uses a dedicated synchronized RNG; flying arms remain local bounded gore.

actor UTNTAltZombieMan : UTNTZombieMan replaces UTNTZombieMan
{
    var int user_altdeathchoice;
    States
    {
    Death:
        "####" "#" 0 A_SetUserVar("user_altdeathchoice",Random[utntaltdeath](0,1))
        "####" "#" 0 A_JumpIf(user_altdeathchoice==0,"Death2")
        Goto Super::Death
    Death2:
        UDZ1 A 8
        UDZ1 B 8 A_Scream
        UDZ1 C 8 A_NoBlocking
        UDZ1 D 8
        UDZ1 E -1
        Stop
    Raise:
        POSS A 0
        Goto Super::Raise
    }
}
